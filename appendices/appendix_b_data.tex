\section{SQLite Database Relational Schema DDL}
\begin{lstlisting}[language=SQL, caption={Relational SQLite Database Table Definitions}]
CREATE TABLE staff (
    staff_id INTEGER PRIMARY KEY AUTOINCREMENT,
    full_name TEXT NOT NULL,
    email TEXT NOT NULL,
    phone_number TEXT NOT NULL,
    department TEXT,
    special_medical_notes TEXT,
    is_active INTEGER DEFAULT 1
);

CREATE TABLE devices (
    device_id INTEGER PRIMARY KEY,
    lora_deveui TEXT UNIQUE,
    ble_mac_address TEXT UNIQUE,
    assigned_staff_id INTEGER,
    battery_percentage REAL,
    last_heartbeat DATETIME,
    FOREIGN KEY(assigned_staff_id) REFERENCES staff(staff_id)
);

CREATE TABLE room_zones (
    zone_id INTEGER PRIMARY KEY AUTOINCREMENT,
    zone_name TEXT NOT NULL,
    x_min REAL, y_min REAL,
    x_max REAL, y_max REAL,
    floor_level INTEGER DEFAULT 1
);

CREATE TABLE incident_logs (
    log_id INTEGER PRIMARY KEY AUTOINCREMENT,
    timestamp DATETIME DEFAULT CURRENT_TIMESTAMP,
    device_id INTEGER,
    event_type TEXT,
    raw_rssi_data TEXT,
    estimated_x REAL, estimated_y REAL,
    resolved_room_name TEXT,
    dispatch_status TEXT,
    is_drill INTEGER DEFAULT 0,
    FOREIGN KEY(device_id) REFERENCES devices(device_id)
);
\end{lstlisting}
